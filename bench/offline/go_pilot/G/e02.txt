package main

import (
	"database/sql"
	"encoding/json"
	"errors"
	"net/http"
	"strconv"

	"platformkit"
)

func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}

func fail(w http.ResponseWriter, status int, name, detail string) {
	writeJSON(w, status, map[string]any{"error": name, "detail": detail})
}

func invalid(w http.ResponseWriter, detail ...string) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusUnprocessableEntity)
	_ = json.NewEncoder(w).Encode(map[string]any{"error": "validation", "detail": detail})
}

func Register(mux *http.ServeMux, k *platformkit.Kit) {
	mux.HandleFunc("POST /orders/{order_id}/pay", func(w http.ResponseWriter, r *http.Request) {
		user, ok := k.Authenticate(w, r, "Customer")
		if !ok {
			return
		}
		id, err := strconv.ParseInt(r.PathValue("order_id"), 10, 64)
		if err != nil || id < 1 {
			invalid(w, "order_id: a positive integer")
			return
		}
		var status string
		err = k.DB.QueryRow(
			"SELECT status FROM orders WHERE id = ? AND (customer_id = ? OR ? = 'Admin')",
			id, user.ID, user.Role,
		).Scan(&status)
		if errors.Is(err, sql.ErrNoRows) {
			fail(w, http.StatusNotFound, "not_found", "order")
			return
		}
		if err != nil {
			platformkit.WriteError(w, err)
			return
		}
		if status != "pending" {
			fail(w, http.StatusConflict, "conflict", "order is not pending")
			return
		}
		if _, err := k.DB.Exec("UPDATE orders SET status = 'paid' WHERE id = ?", id); err != nil {
			platformkit.WriteError(w, err)
			return
		}
		writeJSON(w, http.StatusOK, map[string]any{"id": id, "status": "paid"})
	})
}
